\section{Commands}

\begin{ENTRIES}{1cm}

\IT{\Pair{open}{file}} {Opens \Arg{file} and returns \Ret{\Yes}.}
\index{open@\texttt{open}}%

The \Product{} \Edition{} reads \Many{\Arg{path}} from \Lit{stdin}.

Use \Opt{mode} or \Opt[fast]{mode}, with \Heavy{bold words} and \Mono{mono words}.

Math stays math: \Many{$x_i$} and $\alpha + \beta$.

Layout only: \Layout{spaced} shows its argument, and \Spin{} stays as source.

\ENTRY{\Stack{(\Kw{open} \Arg{
      file})\\
    (\Kw{close} \Arg{file})}{.}}
{
  Opens or closes \Arg{file}. Returns
  \Ret{\Yes} on success.
}

\ENTRY{(\Choice{\Kw{read}\\ \Kw{write}}{\}} \Arg{stream})} {Moves bytes.}

\end{ENTRIES}
